# ---- 3. ОСНОВНОЙ ЗАПРОС: кандидаты --------------------------------------------------------------
# Запуск = устройство обратилось не менее чем в 6 разных сервисов за одну минуту (веер запросов при старте).
# Кандидат = запуск в [from, to), а запросов к trisigma после него нет ни в этом окне, ни в 30 минутах после.
# Запрос к trisigma с любым HTTP-статусом (в т.ч. 401) считается запросом (только 2xx/3xx: README, tri_ok_only).
# Порог версии: строго больше 11.10.0 (для «11.10.0 и выше» замените `> 11010000` на `>= 11010000` в двух местах).
# Результат предварительный: следующим шагом проверьте кандидатов на запросы ПОЗЖЕ и в индексе composition.
POST /_query?format=txt
{
  "query": """
    FROM mpback-k8s-ingress-*
    | WHERE @timestamp >= "2026-09-28T00:00:00Z" AND @timestamp < "2026-09-29T00:30:00Z" AND log.request_hostname == "orderapp.burgerkingrus.ru"
    | GROK log.http_x_burgerking_version "^%{INT:v_major:int}[.]%{INT:v_minor:int}[.]%{INT:v_patch:int}"
    | EVAL ver_num = v_major * 1000000 + v_minor * 1000 + v_patch, is_tri_path = TO_LOWER(log.request_url) LIKE "*trisigma*"
    | WHERE ver_num > 11010000 OR is_tri_path
    | GROK log.http_user_agent "\"id\"%{SPACE}:%{SPACE}\"%{DATA:dev}\""
    | WHERE dev IS NOT NULL
    | EVAL is_tri = is_tri_path, ok = NOT is_tri_path AND TO_LOWER(log.http_x_burgerking_platform) IN ("ios", "android") AND ver_num > 11010000, uid_n = TO_LONG(log.http_x_burgerking_user_id)
    | DISSECT log.request_url "/gateway/%{svc}/%{?rest}"
    | EVAL svc_ok = CASE(ok, svc), ver_ok = CASE(ok, ver_num), plat_ok = CASE(ok, CASE(TO_LOWER(log.http_x_burgerking_platform) == "ios", 4, 5)), tri_flag = CASE(is_tri, 1, 0)
    | EVAL minute = DATE_TRUNC(1 minute, @timestamp)
    | STATS svc_cnt = COUNT_DISTINCT(svc_ok), tri_cnt = SUM(tri_flag), ver_num = MAX(ver_ok), plat = MAX(plat_ok), last_seen = MAX(@timestamp), uid = MAX(uid_n) BY dev, minute
    | EVAL is_launch = svc_cnt >= 6 AND minute < "2026-09-29T00:00:00Z"
    | EVAL launch_flag = CASE(is_launch, 1, 0), launch_min = CASE(is_launch, minute), fan = CASE(is_launch, svc_cnt), tri_min = CASE(tri_cnt > 0, minute)
    | STATS launches = SUM(launch_flag), first_launch = MIN(launch_min), max_fanout = MAX(fan), tri_total = SUM(tri_cnt), last_tri = MAX(tri_min), ver_num = MAX(ver_num), plat = MAX(plat), last_seen = MAX(last_seen), uid = MAX(uid) BY dev
    | WHERE launches > 0 AND (tri_total == 0 OR last_tri < first_launch - 2 minutes)
    | EVAL platform = CASE(plat == 4, "ios", "android"), app_version = CONCAT(TO_STRING(ver_num / 1000000), ".", TO_STRING((ver_num / 1000) % 1000), ".", TO_STRING(ver_num % 1000))
    | KEEP uid, dev, platform, app_version, first_launch, launches, max_fanout, last_seen, tri_total
    | SORT first_launch
    | LIMIT 10000
  """
}
