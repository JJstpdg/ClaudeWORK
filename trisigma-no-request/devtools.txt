# ---- 4. ОСНОВНОЙ ЗАПРОС: кандидаты --------------------------------------------------------------
# Запуск = устройство обратилось не менее чем в 7 разных сервисов за одну минуту (веер запросов при старте).
# Кандидат = запуск в [from, to), а запросов к trisigma после него нет ни в этом окне, ни в 30 минутах после.
# Порог версии: строго больше 11.10.0 (для «11.10.0 и выше» замените `> 11010000` на `>= 11010000`).
# Окно начните с нескольких часов: запрос группирует по устройству и минуте, на суточном окне ему нужно много памяти.
# Результат - предварительный: следующим шагом проверьте кандидатов на запросы ПОЗЖЕ и в индексе composition.
POST /_query?format=txt
{
  "query": """
    FROM mpback-k8s-ingress*
    | WHERE @timestamp >= "2026-09-29T00:00:00Z" AND @timestamp < "2026-09-29T06:30:00Z" AND device_id IS NOT NULL
    | GROK app_version "^%{INT:v_major:int}[.]%{INT:v_minor:int}[.]%{INT:v_patch:int}"
    | EVAL is_tri = TO_LOWER(path) LIKE "*trisigma*", ver_num = v_major * 1000000 + v_minor * 1000 + v_patch
    | EVAL ok = NOT is_tri AND TO_LOWER(platform) IN ("ios", "android") AND ver_num > 11010000
    | DISSECT path "/gateway/%{svc}/%{?rest}"
    | EVAL svc_ok = CASE(ok, svc), ver_ok = CASE(ok, ver_num), plat_ok = CASE(ok, CASE(TO_LOWER(platform) == "ios", 4, 5)), tri_flag = CASE(is_tri, 1, 0)
    | EVAL minute = DATE_TRUNC(1 minute, @timestamp)
    | RENAME device_id AS dev
    | STATS svc_cnt = COUNT_DISTINCT(svc_ok), tri_cnt = SUM(tri_flag), ver_num = MAX(ver_ok), plat = MAX(plat_ok), last_seen = MAX(@timestamp), uid = MAX(user_id) BY dev, minute
    | EVAL is_launch = svc_cnt >= 7 AND minute < "2026-09-29T06:00:00Z"
    | EVAL launch_flag = CASE(is_launch, 1, 0), launch_min = CASE(is_launch, minute), fan = CASE(is_launch, svc_cnt), tri_min = CASE(tri_cnt > 0, minute)
    | STATS launches = SUM(launch_flag), first_launch = MIN(launch_min), max_fanout = MAX(fan), tri_total = SUM(tri_cnt), last_tri = MAX(tri_min), ver_num = MAX(ver_num), plat = MAX(plat), last_seen = MAX(last_seen), uid = MAX(uid) BY dev
    | WHERE launches > 0 AND (tri_total == 0 OR last_tri < first_launch - 2 minutes)
    | EVAL platform = CASE(plat == 4, "ios", "android"), app_version = CONCAT(TO_STRING(ver_num / 1000000), ".", TO_STRING((ver_num / 1000) % 1000), ".", TO_STRING(ver_num % 1000))
    | KEEP uid, dev, platform, app_version, first_launch, launches, max_fanout, last_seen, tri_total
    | SORT first_launch
    | LIMIT 10000
  """
}



# ---- 5. Проверка кандидатов: у кого из них запрос в trisigma всё-таки был (это «ошибочные») -----
# Подставьте device_id из шага 4 (до нескольких сотен за раз) и время начала окна.
# 5a. Ingress: запросы к trisigma БЕЗ верхней границы по времени. Кто вернулся в ответе - тот не наш.

{
  "size": 0,
  "query": {"bool": {"filter": [
    {"terms": {"device_id": ["<device_id_1>", "<device_id_2>"]}},
    {"range": {"@timestamp": {"gte": "2026-09-29T00:00:00Z"}}},
    {"wildcard": {"path": {"value": "*trisigma*", "case_insensitive": true}}}
  ]}},
  "aggs": {"by_device": {
    "terms": {"field": "device_id", "size": 1000},
    "aggs": {"first_request": {"min": {"field": "@timestamp"}}}
  }}
}
